-- KRAKESMP OWNER SECURITY alter table public.ks_players enable row level security; alter table public.ks_rounds enable row level security; -- Öffentliche Rangliste drop policy if exists "Leaderboard is public" on public.ks_players; create policy "Leaderboard is public" on public.ks_players for select to anon, authenticated using (true); -- Spieler dürfen nur eigene Spielrunden lesen drop policy if exists "Players see own rounds" on public.ks_rounds; create policy "Players see own rounds" on public.ks_rounds for select to authenticated using (user_id = (select auth.uid())); -- Owner darf alle Spielrunden lesen create policy "Owner reads all rounds" on public.ks_rounds for select to authenticated using ((select public.krake_is_admin())); -- Owner darf Spielrunden eintragen create policy "Owner inserts rounds" on public.ks_rounds for insert to authenticated with check ((select public.krake_is_admin())); -- Owner darf Spielerstatistiken bearbeiten create policy "Owner updates players" on public.ks_players for update to authenticated using ((select public.krake_is_admin())) with check ((select public.krake_is_admin()));
